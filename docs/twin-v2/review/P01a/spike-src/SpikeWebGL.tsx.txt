"use client";

/**
 * P01a renderer spike, WebGL2 path: WebGLRenderer, CSM (3 cascades), GTAO, bloom, AgX output.
 * Throwaway. See shared.ts.
 */
import { useEffect, useRef } from "react";
import { Canvas, useFrame, useThree } from "@react-three/fiber";
import { CameraControls, Html } from "@react-three/drei";
import * as THREE from "three";
import { CSM } from "three/examples/jsm/csm/CSM.js";
import { EffectComposer } from "three/examples/jsm/postprocessing/EffectComposer.js";
import { RenderPass } from "three/examples/jsm/postprocessing/RenderPass.js";
import { GTAOPass } from "three/examples/jsm/postprocessing/GTAOPass.js";
import { UnrealBloomPass } from "three/examples/jsm/postprocessing/UnrealBloomPass.js";
import { OutputPass } from "three/examples/jsm/postprocessing/OutputPass.js";
import {
  BLOOM,
  CAMERA_POS,
  CAMERA_TARGET,
  SHADOW_FAR,
  SUN_DIR,
  buildCrowd,
  buildPointLights,
  countScene,
  loadAssets,
  readOptions,
  type GroundTextures,
  type SpikeStats,
} from "./shared";

function groundMaterial(g: GroundTextures) {
  const mat = new THREE.MeshStandardMaterial({ roughness: 1, metalness: 0 });
  // (a normal map must be set so three builds the tangent frame and passes the uv through)
  mat.normalMap = g.nor[0];
  const patch = (shader: THREE.WebGLProgramParametersWithUniforms) => {
    for (let i = 0; i < 4; i++) {
      shader.uniforms[`gD${i}`] = { value: g.diff[i] };
      shader.uniforms[`gN${i}`] = { value: g.nor[i] };
      shader.uniforms[`gR${i}`] = { value: g.rough[i] };
    }
    shader.vertexShader = shader.vertexShader
      .replace("#include <common>", "#include <common>\nattribute vec4 layerW;\nvarying vec4 vLayerW;")
      .replace("#include <begin_vertex>", "#include <begin_vertex>\nvLayerW = layerW;");
    const decl = [0, 1, 2, 3].map((i) => `uniform sampler2D gD${i}; uniform sampler2D gN${i}; uniform sampler2D gR${i};`).join("\n");
    const mix = (k: string) => [0, 1, 2, 3].map((i) => `texture2D( g${k}${i}, vNormalMapUv ) * vLayerW[${i}]`).join(" + ");
    shader.fragmentShader = shader.fragmentShader
      .replace("#include <common>", `#include <common>\nvarying vec4 vLayerW;\n${decl}`)
      .replace("#include <map_fragment>", `diffuseColor.rgb *= ( ${mix("D")} ).rgb;`)
      .replace("#include <roughnessmap_fragment>", `float roughnessFactor = roughness * ( ${mix("R")} ).g;`)
      .replace("#include <normal_fragment_maps>", `vec3 mapN = ( ${mix("N")} ).xyz * 2.0 - 1.0;\nnormal = normalize( tbn * mapN );`);
  };
  return { mat, patch };
}

function Scene() {
  const { gl, scene, camera, size, get } = useThree();
  const run = useRef<{ frame: (dt: number) => void } | null>(null);

  useEffect(() => {
    const opts = readOptions();
    const stats: SpikeStats = {
      path: "webgl",
      backend: "WebGLRenderer (WebGL2)",
      ready: false,
      timings: {},
      checks: {},
      info: () => ({ calls: gl.info.render.calls, triangles: gl.info.render.triangles, programs: gl.info.programs?.length, geometries: gl.info.memory.geometries, textures: gl.info.memory.textures }),
    };
    window.__SPIKE__ = stats;
    let disposed = false;
    const cleanup: (() => void)[] = [];

    (async () => {
      const assets = await loadAssets(opts);
      if (disposed) return;
      stats.timings.loadMs = Math.round(assets.loadMs);

      gl.toneMapping = THREE.AgXToneMapping;
      gl.toneMappingExposure = 1;
      gl.shadowMap.enabled = opts.shadows;
      gl.shadowMap.type = THREE.PCFSoftShadowMap;
      gl.info.autoReset = false;
      // (the composer renders the scene twice, colour then normals for GTAO: draw the shadow maps once a frame, not once a pass)
      gl.shadowMap.autoUpdate = false;
      scene.background = new THREE.Color(0x9fc4e8);

      const cam = camera as THREE.PerspectiveCamera;
      cam.position.set(...CAMERA_POS);
      cam.lookAt(...CAMERA_TARGET);
      cam.updateMatrixWorld();
      const controls = get().controls as unknown as { setLookAt?: (...a: (number | boolean)[]) => void } | null;
      controls?.setLookAt?.(...CAMERA_POS, ...CAMERA_TARGET, false);
      stats.checks.cameraControls = !!controls?.setLookAt;

      const root = new THREE.Group();
      scene.add(root);
      cleanup.push(() => scene.remove(root));
      root.add(new THREE.HemisphereLight(0xbfd8ff, 0x5a4a38, 0.9));

      let csm: CSM | null = null;
      if (opts.shadows) {
        csm = new CSM({ camera: cam, parent: root, cascades: 3, maxFar: SHADOW_FAR, mode: "practical", shadowMapSize: opts.shadowMap, lightDirection: SUN_DIR.clone(), lightIntensity: 3, lightNear: 1, lightFar: 1200, lightMargin: 200, shadowBias: -0.0002 });
        cleanup.push(() => {
          csm?.remove();
          csm?.dispose();
        });
      } else {
        const sun = new THREE.DirectionalLight(0xffffff, 3);
        sun.position.copy(SUN_DIR).multiplyScalar(-300);
        root.add(sun);
      }
      const withCsm = (m: THREE.Material, extra?: (s: THREE.WebGLProgramParametersWithUniforms) => void) => {
        if (csm) csm.setupMaterial(m);
        const inner = m.onBeforeCompile;
        m.onBeforeCompile = (shader, renderer) => {
          if (csm) inner.call(m, shader, renderer);
          extra?.(shader);
        };
        m.customProgramCacheKey = () => (extra ? "spike-ground" : "spike-csm");
      };

      const ground = groundMaterial(assets.ground);
      withCsm(ground.mat, ground.patch);
      const terrain = new THREE.Mesh(assets.terrain, ground.mat);
      terrain.receiveShadow = true;
      root.add(terrain);

      for (const p of assets.props) {
        withCsm(p.material);
        const im = new THREE.InstancedMesh(p.geometry, p.material, opts.props);
        (im.instanceMatrix.array as Float32Array).set(p.matrices as Float32Array);
        im.instanceMatrix.needsUpdate = true;
        im.castShadow = true;
        im.receiveShadow = true;
        im.computeBoundingSphere();
        root.add(im);
      }

      const crowd = buildCrowd(assets, opts.chars);
      const seen = new Set<THREE.Material>();
      crowd.group.traverse((o) => {
        const m = (o as THREE.Mesh).material as THREE.Material | undefined;
        if (m && !seen.has(m)) {
          seen.add(m);
          withCsm(m);
        }
      });
      root.add(crowd.group);
      root.add(buildPointLights(opts.lights));

      const target = new THREE.WebGLRenderTarget(1, 1, { type: THREE.HalfFloatType, ...(opts.ao && opts.depthOnlyAo ? { depthTexture: new THREE.DepthTexture(1, 1) } : {}) });
      const composer = new EffectComposer(gl, target);
      composer.setPixelRatio(gl.getPixelRatio());
      composer.setSize(size.width, size.height);
      composer.addPass(new RenderPass(scene, cam));
      if (opts.ao) {
        const ao = new GTAOPass(scene, cam, size.width, size.height);
        ao.output = GTAOPass.OUTPUT.Default;
        // (RenderPass draws into the composer's read buffer, which is renderTarget2 for the first pass)
        if (opts.depthOnlyAo) ao.setGBuffer(composer.renderTarget2.depthTexture as THREE.DepthTexture);
        stats.checks.aoDepthOnly = opts.depthOnlyAo;
        composer.addPass(ao);
      }
      if (opts.bloom) composer.addPass(new UnrealBloomPass(new THREE.Vector2(size.width, size.height), BLOOM.strength, BLOOM.radius, BLOOM.threshold));
      composer.addPass(new OutputPass());
      cleanup.push(() => composer.dispose());

      stats.scene = { ...countScene(scene), charHeight: crowd.sourceHeight, clip: crowd.clip };
      const buf = gl.getDrawingBufferSize(new THREE.Vector2());
      stats.canvas = [buf.x, buf.y];

      const frame = (dt: number) => {
        for (const m of crowd.mixers) m.update(dt);
        csm?.update();
        gl.info.reset();
        gl.shadowMap.needsUpdate = true;
        composer.render(dt);
      };

      const c0 = performance.now();
      await gl.compileAsync(scene, cam);
      stats.timings.compileMs = Math.round(performance.now() - c0);
      const r0 = performance.now();
      frame(0);
      gl.getContext().finish();
      stats.timings.firstRenderMs = Math.round(performance.now() - r0);
      await new Promise((r) => requestAnimationFrame(() => r(null)));
      stats.timings.navToFirstFrameMs = Math.round(performance.now());
      if (disposed) return;
      run.current = { frame };
      stats.ready = true;
    })().catch((e) => {
      stats.error = String(e?.stack || e).slice(0, 600);
      console.error(e);
    });

    return () => {
      disposed = true;
      run.current = null;
      cleanup.forEach((f) => f());
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  useFrame((_, dt) => run.current?.frame(dt), 1);
  return null;
}

export default function SpikeWebGL() {
  return (
    <Canvas
      shadows
      dpr={[1, 2]}
      camera={{ fov: 45, near: 0.5, far: 1500, position: CAMERA_POS }}
      gl={{ antialias: false, powerPreference: "high-performance", stencil: false }}
    >
      <Scene />
      <CameraControls makeDefault />
      <Html position={[6, 14, -4]} center>
        <div data-spike-label className="rounded bg-black/70 px-2 py-1 font-mono text-[11px] text-white whitespace-nowrap">
          Html label · webgl
        </div>
      </Html>
    </Canvas>
  );
}
